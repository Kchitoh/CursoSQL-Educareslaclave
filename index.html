Sublenguaje DDL (Data Definition Language)

CREATE TABLE alumnos (
  id INT PRIMARY KEY AUTO_INCREMENT,
  nombre VARCHAR(50) NOT NULL
);
ALTER TABLE alumnos ADD COLUMN email VARCHAR(100);
DROP TABLE alumnos;
TRUNCATE TABLE alumnos;

Sublenguaje DML (Data Manipulation Language)

INSERT INTO alumnos (nombre) VALUES ('Nicolás');
UPDATE alumnos SET nombre = 'Nico' WHERE id = 1;
DELETE FROM alumnos WHERE id = 1;
SELECT a.nombre, c.curso
FROM alumnos a
INNER JOIN inscripciones c ON a.id = c.alumno_id
WHERE a.estado = 1;
